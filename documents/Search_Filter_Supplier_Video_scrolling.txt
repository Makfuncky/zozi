```python
# backend/services/advanced_filter_service.py

from typing import Dict, List, Any, Optional
from sqlalchemy import func, and_, or_
from sqlalchemy.orm import Session
import json

class AdvancedFilterService:
    def __init__(self, db: Session):
        self.db = db
    
    def get_available_filters(self, category_id: int, search_query: str = None) -> Dict[str, Any]:
        """
        Get all available filters with counts for a category
        """
        base_query = self.db.query(Product).filter(
            Product.category_id == category_id,
            Product.is_active == True
        )
        
        if search_query:
            base_query = base_query.filter(
                Product.search_vector.match(search_query, postgresql_regconfig='english')
            )
        
        filters = {
            'price_range': self._get_price_range_stats(base_query),
            'brands': self._get_brands_with_counts(base_query),
            'ratings': self._get_rating_distribution(base_query),
            'attributes': self._get_dynamic_attributes(base_query, category_id),
            'availability': self._get_availability_stats(base_query),
            'shipping': self._get_shipping_options(base_query)
        }
        
        return filters
    
    def apply_filters(self, query, filters: Dict[str, Any]) -> Query:
        """
        Apply multiple filters to a query
        """
        # Price range
        if 'min_price' in filters:
            query = query.filter(Product.price >= filters['min_price'])
        if 'max_price' in filters:
            query = query.filter(Product.price <= filters['max_price'])
        
        # Brand filter
        if 'brands' in filters and filters['brands']:
            query = query.filter(Product.brand.in_(filters['brands']))
        
        # Rating filter
        if 'min_rating' in filters:
            query = query.filter(Product.average_rating >= filters['min_rating'])
        
        # Dynamic attributes (JSONB)
        if 'attributes' in filters:
            for attr_key, attr_values in filters['attributes'].items():
                if attr_values:
                    query = query.filter(
                        Product.filter_attributes.has_key(attr_key)
                    )
                    query = query.filter(
                        Product.filter_attributes[attr_key].cast(String).in_(attr_values)
                    )
        
        # Availability
        if 'in_stock' in filters and filters['in_stock']:
            query = query.filter(Product.stock_quantity > 0)
        
        return query
    
    def _get_price_range_stats(self, query) -> Dict[str, float]:
        result = query.with_entities(
            func.min(Product.price),
            func.max(Product.price),
            func.avg(Product.price)
        ).first()
        
        return {
            'min': float(result[0]) if result[0] else 0,
            'max': float(result[1]) if result[1] else 0,
            'avg': float(result[2]) if result[2] else 0
        }
    
    def _get_brands_with_counts(self, query) -> List[Dict[str, Any]]:
        return query.with_entities(
            Product.brand,
            func.count(Product.id).label('count')
        ).group_by(Product.brand).order_by(func.count(Product.id).desc()).all()
    
    def _get_dynamic_attributes(self, query, category_id: int) -> List[Dict[str, Any]]:
        # Get filter metadata for category
        metadata = self.db.query(ProductFilterMetadata).filter(
            ProductFilterMetadata.category_id == category_id,
            ProductFilterMetadata.is_active == True
        ).all()
        
        filters_with_counts = []
        for meta in metadata:
            options = self.db.query(ProductFilterOption).filter(
                ProductFilterOption.filter_metadata_id == meta.id
            ).order_by(ProductFilterOption.sort_order).all()
            
            filters_with_counts.append({
                'id': meta.id,
                'name': meta.filter_name,
                'type': meta.filter_type,
                'options': [
                    {
                        'value': opt.option_value,
                        'display': opt.option_display_name,
                        'count': opt.product_count
                    }
                    for opt in options
                ]
            })
        
        return filters_with_counts
```

```python
    in_stock: Optional[bool] = None
    sort_by: str = "relevance"  # relevance, price_asc, price_desc, rating, newest
    page: int = 1
    page_size: int = 20

@router.get("/search/filters")
async def get_available_filters(
    category_id: int = Query(...),
    search_query: Optional[str] = Query(None)
):
    """
    Get all available filters with product counts
    """
    filter_service = AdvancedFilterService(db)
    filters = filter_service.get_available_filters(category_id, search_query)
    return {"filters": filters}

@router.post("/search/advanced")
async def advanced_search(filters: FilterRequest):
    """
    Advanced search with multiple filters
    """
    filter_service = AdvancedFilterService(db)
    
    # Base query
    query = db.query(Product).filter(Product.is_active == True)
    
    # Apply category filter
    if filters.category_id:
        query = query.filter(Product.category_id == filters.category_id)
    
    # Apply text search
    if filters.search_query:
        query = query.filter(
            Product.search_vector.match(filters.search_query, postgresql_regconfig='english')
        )
    
    # Apply all filters
    query = filter_service.apply_filters(query, filters.dict())
    
    # Apply sorting
    if filters.sort_by == "price_asc":
        query = query.order_by(Product.price.asc())
    elif filters.sort_by == "price_desc":
        query = query.order_by(Product.price.desc())
    elif filters.sort_by == "rating":
        query = query.order_by(Product.average_rating.desc())
    elif filters.sort_by == "newest":
        query = query.order_by(Product.created_at.desc())
    
    # Pagination
    total = query.count()
    products = query.offset((filters.page - 1) * filters.page_size).limit(filters.page_size).all()
    
    return {
        "products": products,
        "total": total,
        "page": filters.page,
        "page_size": filters.page_size,
        "total_pages": (total + filters.page_size - 1) // filters.page_size
    }
```

```typescript
// frontend/components/AdvancedFilter.tsx

import React, { useState, useEffect } from 'react';

interface FilterState {
  minPrice: number;
  maxPrice: number;
  brands: string[];
  minRating: number;
  attributes: Record<string, string[]>;
  inStock: boolean;
}

export const AdvancedFilter: React.FC<{
  categoryId: number;
  onFilterChange: (filters: FilterState) => void;
}> = ({ categoryId, onFilterChange }) => {
  const [filters, setFilters] = useState<FilterState>({
    minPrice: 0,
    maxPrice: 10000,
    brands: [],
    minRating: 0,
    attributes: {},
    inStock: false
  });
  
  const [availableFilters, setAvailableFilters] = useState<any>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchAvailableFilters();
  }, [categoryId]);

  const fetchAvailableFilters = async () => {
    setLoading(true);
    const response = await fetch(`/api/search/filters?category_id=${categoryId}`);
    const data = await response.json();
    setAvailableFilters(data.filters);
    
    // Initialize price range
    if (data.filters.price_range) {
      setFilters(prev => ({
        ...prev,
        minPrice: data.filters.price_range.min,
        maxPrice: data.filters.price_range.max
      }));
    }
    setLoading(false);
  };

  const handlePriceChange = (min: number, max: number) => {
    const newFilters = { ...filters, minPrice: min, maxPrice: max };
    setFilters(newFilters);
    onFilterChange(newFilters);
  };

  const handleBrandToggle = (brand: string) => {
    const newBrands = filters.brands.includes(brand)
      ? filters.brands.filter(b => b !== brand)
      : [...filters.brands, brand];
    
    const newFilters = { ...filters, brands: newBrands };
    setFilters(newFilters);
    onFilterChange(newFilters);
  };

  const handleAttributeChange = (attrKey: string, value: string) => {
    const currentValues = filters.attributes[attrKey] || [];
    const newValues = currentValues.includes(value)
      ? currentValues.filter(v => v !== value)
      : [...currentValues, value];
    
    const newAttributes = { ...filters.attributes, [attrKey]: newValues };
    const newFilters = { ...filters, attributes: newAttributes };
    setFilters(newFilters);
    onFilterChange(newFilters);
  };

  if (loading) return <div>Loading filters...</div>;

  return (
    <div className="space-y-6">
      {/* Price Range Slider */}
      {availableFilters?.price_range && (
        <div className="filter-section">
          <h3 className="font-semibold mb-3">Price Range</h3>
          <input
            type="range"
            min={availableFilters.price_range.min}
            max={availableFilters.price_range.max}
            value={filters.maxPrice}
            onChange={(e) => handlePriceChange(filters.minPrice, parseInt(e.target.value))}
            className="w-full"
          />
          <div className="flex justify-between text-sm text-gray-600">
            <span>${filters.minPrice}</span>
            <span>${filters.maxPrice}</span>
          </div>
        </div>
      )}

      {/* Brand Filter */}
      {availableFilters?.brands && (
        <div className="filter-section">
          <h3 className="font-semibold mb-3">Brands</h3>
          <div className="space-y-2">
            {availableFilters.brands.map((brand: any) => (
              <label key={brand.brand} className="flex items-center">
                <input
                  type="checkbox"
                  checked={filters.brands.includes(brand.brand)}
                  onChange={() => handleBrandToggle(brand.brand)}
                  className="mr-2"
                />
                <span>{brand.brand}</span>
                <span className="ml-auto text-gray-500 text-sm">({brand.count})</span>
              </label>
            ))}
          </div>
        </div>
      )}

      {/* Rating Filter */}
      <div className="filter-section">
        <h3 className="font-semibold mb-3">Minimum Rating</h3>
        {[4, 3, 2, 1].map((rating) => (
          <label key={rating} className="flex items-center mb-2">
            <input
              type="radio"
              name="rating"
              checked={filters.minRating === rating}
              onChange={() => {
                const newFilters = { ...filters, minRating: rating };
                setFilters(newFilters);
                onFilterChange(newFilters);
              }}
              className="mr-2"
            />
            <span>{'★'.repeat(rating)} & Up</span>
          </label>
        ))}
      </div>

      {/* Dynamic Attributes */}
      {availableFilters?.attributes?.map((attr: any) => (
        <div key={attr.id} className="filter-section">
          <h3 className="font-semibold mb-3">{attr.name}</h3>
          <div className="space-y-2">
            {attr.options.map((option: any) => (
              <label key={option.value} className="flex items-center">
                <input
                  type="checkbox"
                  checked={(filters.attributes[attr.name] || []).includes(option.value)}
                  onChange={() => handleAttributeChange(attr.name, option.value)}
                  className="mr-2"
                />
                <span>{option.display}</span>
                <span className="ml-auto text-gray-500 text-sm">({option.count})</span>
              </label>
            ))}
          </div>
        </div>
      ))}

      {/* In Stock Filter */}
      <div className="filter-section">
        <label className="flex items-center">
          <input
            type="checkbox"
            checked={filters.inStock}
            onChange={(e) => {
              const newFilters = { ...filters, inStock: e.target.checked };
              setFilters(newFilters);
              onFilterChange(newFilters);
            }}
            className="mr-2"
          />
          <span>In Stock Only</span>
        </label>
      </div>
    </div>
  );
```

```python
# backend/services/search_engine.py

from elasticsearch import Elasticsearch
from elasticsearch_dsl import Search, Q
import re
from typing import List, Dict, Any

class AdvancedSearchEngine:
    def __init__(self):
        self.es = Elasticsearch([{'host': 'localhost', 'port': 9200}])
        self.index_name = 'products'
    
    def parse_query(self, query: str) -> Dict[str, Any]:
        """
        Parse natural language query into structured filters
        Examples:
        - "juicer under $100" -> {term: "juicer", max_price: 100}
        - "red dress size M" -> {term: "dress", color: "red", size: "M"}
        - "laptop with 16GB RAM" -> {term: "laptop", ram: "16GB"}
        """
        parsed = {
            'term': query,
            'filters': {}
        }
        
        # Price extraction
        price_match = re.search(r'under \$?(\d+)|below \$?(\d+)|less than \$?(\d+)', query, re.IGNORECASE)
        if price_match:
            price = price_match.group(1) or price_match.group(2) or price_match.group(3)
            parsed['filters']['max_price'] = int(price)
            parsed['term'] = re.sub(r'under \$?\d+|below \$?\d+|less than \$?\d+', '', query, flags=re.IGNORECASE).strip()
        
        # Rating extraction
        rating_match = re.search(r'(\d)\+? stars?|(\d)\+? rating', query, re.IGNORECASE)
        if rating_match:
            rating = rating_match.group(1) or rating_match.group(2)
            parsed['filters']['min_rating'] = int(rating)
        
        # Brand extraction (common brands)
        brands = ['samsung', 'apple', 'sony', 'lg', 'nike', 'adidas']
        for brand in brands:
            if brand in query.lower():
                parsed['filters']['brand'] = brand.capitalize()
                parsed['term'] = query.replace(brand, '', flags=re.IGNORECASE).strip()
        
        return parsed
    
    def expand_query(self, query: str) -> List[str]:
        """
        Expand query with synonyms and related terms
        """
        synonyms = {
            'juicer': ['blender', 'mixer', 'extractor'],
            'laptop': ['notebook', 'computer', 'pc'],
            'dress': ['gown', 'outfit', 'clothing'],
            'shoes': ['footwear', 'sneakers', 'boots']
        }
        
        expanded = [query]
        words = query.lower().split()
        
        for word in words:
            if word in synonyms:
                for synonym in synonyms[word]:
                    expanded_query = query.lower().replace(word, synonym)
                    expanded.append(expanded_query)
        
        return list(set(expanded))
    
    def search(self, query: str, filters: Dict[str, Any] = None, page: int = 1, page_size: int = 20) -> Dict[str, Any]:
        """
        Advanced search with multiple strategies
        """
        parsed = self.parse_query(query)
        expanded_queries = self.expand_query(parsed['term'])
        
        # Build search body
        search_body = {
            "query": {
                "bool": {
                    "should": [],
                    "filter": []
                }
            },
            "aggs": {
                "brands": {
                    "terms": {"field": "brand.keyword", "size": 20}
                },
                "price_ranges": {
                    "range": {
                        "field": "price",
                        "ranges": [
                            {"to": 50},
                            {"from": 50, "to": 100},
                            {"from": 100, "to": 500},
                            {"from": 500}
                        ]
                    }
                },
                "avg_rating": {
                    "avg": {"field": "average_rating"}
                }
            },
            "from": (page - 1) * page_size,
            "size": page_size,
            "highlight": {
                "fields": {
                    "name": {},
                    "description": {},
                    "tags": {}
                }
            }
        }
        
        # Add main query with boosting
        for i, exp_query in enumerate(expanded_queries):
            boost = 3 - (i * 0.5)  # Decrease boost for later expansions
            search_body["query"]["bool"]["should"].append({
                "multi_match": {
                    "query": exp_query,
                    "fields": [
                        "name^3",
                        "description^2",
                        "tags",
                        "brand^2",
                        "category"
                    ],
                    "fuzziness": "AUTO",
                    "boost": boost
                }
            })
        
        # Apply filters
        if filters:
            if 'min_price' in filters:
                search_body["query"]["bool"]["filter"].append({
                    "range": {"price": {"gte": filters['min_price']}}
                })
            if 'max_price' in filters:
                search_body["query"]["bool"]["filter"].append({
                    "range": {"price": {"lte": filters['max_price']}}
                })
            if 'min_rating' in filters:
                search_body["query"]["bool"]["filter"].append({
                    "range": {"average_rating": {"gte": filters['min_rating']}}
                })
            if 'brands' in filters and filters['brands']:
                search_body["query"]["bool"]["filter"].append({
                    "terms": {"brand.keyword": filters['brands']}
                })
        
        # Execute search
        response = self.es.search(index=self.index_name, body=search_body)
        
        return {
            "hits": response['hits']['hits'],
            "total": response['hits']['total']['value'],
            "aggregations": response['aggregations'],
            "took": response['took']
        }
    
    def autocomplete(self, query: str, size: int = 10) -> List[str]:
        """
        Search autocomplete suggestions
        """
        response = self.es.search(
            index=self.index_name,
            body={
                "suggest": {
                    "product-suggest": {
                        "prefix": query,
                        "completion": {
                            "field": "suggest",
                            "size": size,
                            "skip_duplicates": True
                        }
                    }
                }
            }
        )
        
        suggestions = response['suggest']['product-suggest'][0]['options']
        return [s['text'] for s in suggestions]
```

#### **Phase 2: AI-Powered Query Understanding**

```python
# backend/services/ai_search_service.py

import spacy
from typing import Dict, Any, List

class AISearchService:
    def __init__(self):
        self.nlp = spacy.load("en_core_web_sm")
        # Load product catalog for entity recognition
        self.product_entities = self._load_product_entities()
    
    def _load_product_entities(self) -> Dict[str, List[str]]:
        """
        Load product catalog for NER
        """
        return {
            'brands': ['Samsung', 'Apple', 'Sony', 'LG', 'Nike', 'Adidas'],
            'categories': ['Electronics', 'Clothing', 'Home', 'Sports'],
            'attributes': ['color', 'size', 'material', 'weight']
        }
    
    def extract_intent(self, query: str) -> Dict[str, Any]:
        """
        Extract search intent using NLP
        """
        doc = self.nlp(query)
        
        intent = {
            'primary_intent': 'product_search',
            'entities': {
                'brands': [],
                'categories': [],
                'attributes': {},
                'price_range': None,
                'rating': None
            },
            'modifiers': [],
            'sentiment': 'neutral'
        }
        
        # Extract entities
        for ent in doc.ents:
            if ent.label_ == 'PRODUCT':
                # Check if it's a brand
                if ent.text in self.product_entities['brands']:
                    intent['entities']['brands'].append(ent.text)
                else:
                    intent['entities']['categories'].append(ent.text)
            
            elif ent.label_ == 'MONEY':
                intent['entities']['price_range'] = ent.text
            
            elif ent.label_ == 'CARDINAL':
                # Could be rating or quantity
                if 'star' in query.lower() or 'rating' in query.lower():
                    intent['entities']['rating'] = int(ent.text)
        
        # Extract attributes
        for token in doc:
            if token.pos_ == 'ADJ':
                intent['modifiers'].append(token.lemma_)
            elif token.pos_ == 'NOUN' and token.lemma_ in self.product_entities['attributes']:
                # Look for value after attribute
                if token.i + 1 < len(doc):
                    intent['entities']['attributes'][token.lemma_] = doc[token.i + 1].text
        
        # Detect intent type
        if 'compare' in query.lower() or 'vs' in query.lower():
            intent['primary_intent'] = 'compare_products'
        elif 'best' in query.lower() or 'top' in query.lower():
            intent['primary_intent'] = 'recommendation'
        elif 'review' in query.lower() or 'rating' in query.lower():
            intent['primary_intent'] = 'reviews'
        
        return intent
    
    def generate_search_query(self, intent: Dict[str, Any]) -> str:
        """
        Generate optimized search query from intent
        """
        query_parts = []
        
        # Add categories
        if intent['entities']['categories']:
            query_parts.extend(intent['entities']['categories'])
        
        # Add brands
        if intent['entities']['brands']:
            query_parts.extend(intent['entities']['brands'])
        
        # Add modifiers
        if intent['modifiers']:
            query_parts.extend(intent['modifiers'])
        
        return ' '.join(query_parts)
```

```python
# backend/services/video_service.py

import boto3
from botocore.client import Config
from typing import Optional, Dict, Any
import ffmpeg
import os

class VideoProcessingService:
    def __init__(self):
        self.s3_client = boto3.client('s3')
        self.bucket_name = 'zozi-product-videos'
        self.cf_distribution = 'YOUR_CLOUDFRONT_ID'
    
    def upload_video(self, video_file, product_id: int, video_type: str) -> Dict[str, Any]:
        """
        Upload video to S3 and trigger processing
        """
        # Generate unique filename
        file_extension = os.path.splitext(video_file.filename)[1]
        video_key = f"products/{product_id}/original_{uuid.uuid4()}{file_extension}"
        
        # Upload to S3
        self.s3_client.upload_fileobj(
            video_file,
            self.bucket_name,
            video_key,
            ExtraArgs={
                'ContentType': video_file.content_type,
                'ACL': 'private'
            }
        )
        
        # Create video record
        video_record = {
            'product_id': product_id,
            'video_url': f"https://{self.cf_distribution}.cloudfront.net/{video_key}",
            'video_type': video_type,
            'upload_status': 'processing'
        }
        
        # Trigger background processing
        from celery import Celery
        celery_app = Celery('video_processor')
        celery_app.send_task('process_video', args=[video_record['id'], video_key])
        
        return video_record
    
    def generate_thumbnails(self, video_path: str, output_path: str) -> List[str]:
        """
        Generate multiple thumbnails from video
        """
        thumbnails = []
        
        # Generate thumbnail at 10%, 50%, 90% of video
        for percentage in [10, 50, 90]:
            thumb_path = f"{output_path}_{percentage}.jpg"
            (
                ffmpeg
                .input(video_path, ss=f"{percentage}%")
                .output(thumb_path, vframes=1, qscale=2)
                .run(capture_stdout=True, capture_stderr=True)
            )
            thumbnails.append(thumb_path)
        
        return thumbnails
    
    def optimize_video(self, input_path: str, output_path: str, max_duration: int = 30):
        """
        Optimize video for web:
        - Compress to H.264
        - Limit duration to 30s
        - Reduce resolution to 720p max
        - Optimize for mobile
        """
        (
            ffmpeg
            .input(input_path)
            .filter('trim', duration=max_duration)
            .filter('scale', "min(1280,iw)", "min(720,ih)", force_original_aspect_ratio='decrease')
            .output(
                output_path,
                vcodec='libx264',
                acodec='aac',
                preset='medium',
                crf=23,
                movflags='faststart',
                pix_fmt='yuv420p'
            )
            .run(capture_stdout=True, capture_stderr=True)
        )
    
    def get_video_recommendations(self, product_id: int, limit: int = 5) -> List[Dict[str, Any]]:
        """
        Get video recommendations for sidebar
        Strategy:
        1. Videos from same product
        2. Videos from same category
        3. Videos from similar products (based on tags/attributes)
        """
        db = SessionLocal()
        
        # Get current product
        product = db.query(Product).filter(Product.id == product_id).first()
        
        # Query for recommendations
        recommendations = db.query(ProductVideos).join(Product).filter(
            or_(
                Product.category_id == product.category_id,
                Product.tags.overlap(product.tags),
                Product.brand == product.brand
            ),
            ProductVideos.id != product_id,
            ProductVideos.upload_status == 'ready',
            ProductVideos.is_featured == True
        ).order_by(
            ProductVideos.views_count.desc(),
            ProductVideos.created_at.desc()
        ).limit(limit).all()
        
        return [self._video_to_dict(v) for v in recommendations]
    
    def _video_to_dict(self, video: ProductVideos) -> Dict[str, Any]:
        return {
            'id': video.id,
            'product_id': video.product_id,
            'video_url': video.video_url,
            'thumbnail_url': video.thumbnail_url,
            'duration': video.duration,
            'video_type': video.video_type,
            'title': video.title,
            'views_count': video.views_count,
            'product_name': video.product.name if video.product else None,
            'product_price': video.product.price if video.product else None
        }
```

```python
# backend/routers/videos.py

from fastapi import APIRouter, UploadFile, File, Form, HTTPException, Depends
from typing import Optional, List
from pydantic import BaseModel

router = APIRouter()

class VideoCreateRequest(BaseModel):
    product_id: int
    video_type: str
    title: Optional[str] = None
    description: Optional[str] = None

@router.post("/videos/upload")
async def upload_video(
    video: UploadFile = File(...),
    product_id: int = Form(...),
    video_type: str = Form(...),
    title: Optional[str] = Form(None),
    description: Optional[str] = Form(None)
):
    """
    Upload product video (for suppliers)
    """
    # Validate file type
    if not video.content_type.startswith('video/'):
        raise HTTPException(status_code=400, detail="File must be a video")
    
    # Validate file size (max 100MB)
    video.file.seek(0, 2)  # Seek to end
    file_size = video.file.tell()
    video.file.seek(0)  # Seek back to start
    
    if file_size > 100 * 1024 * 1024:  # 100MB
        raise HTTPException(status_code=400, detail="Video size must be less than 100MB")
    
    video_service = VideoProcessingService()
    video_record = video_service.upload_video(video, product_id, video_type)
    
    return {
        "message": "Video uploaded successfully",
        "video_id": video_record['id'],
        "status": "processing"
    }

@router.get("/products/{product_id}/videos")
async def get_product_videos(product_id: int):

    """

    ).order_by(ProductVideos.is_featured.desc(), ProductVideos.created_at.desc()).all()
    
    return {
        "videos": [video_service._video_to_dict(v) for v in videos]
    }

@router.get("/videos/recommendations/{product_id}")

    }

@router.post("/videos/{video_id}/track")
async def track_video_event(
    video_id: int,
    event_type: str,
    watch_duration: Optional[int] = None,
    user_id: Optional[int] = None
):
    """
    Track video analytics (view, pause, complete, etc.)
    """
    db = SessionLocal()
    
    analytics = VideoAnalytics(
        video_id=video_id,
        user_id=user_id,
        event_type=event_type,
        watch_duration=watch_duration
    )
    
    db.add(analytics)
    
    # Update view count
    if event_type == 'view':
        db.query(ProductVideos).filter(ProductVideos.id == video_id).update(
            {"views_count": ProductVideos.views_count + 1}
        )
    
    db.commit()
    
    return {"status": "tracked"}
```

```typescript
// frontend/components/VideoRecommendationSidebar.tsx

import React, { useState, useEffect, useRef } from 'react';

interface Video {
  id: number;
  product_id: number;
  video_url: string;
  thumbnail_url: string;
  duration: number;
  video_type: string;
  title: string;
  views_count: number;
  product_name: string;
  product_price: number;
}

export const VideoRecommendationSidebar: React.FC<{
  productId: number;
  position?: 'right' | 'left';
}> = ({ productId, position = 'right' }) => {
  const [videos, setVideos] = useState<Video[]>([]);
  const [loading, setLoading] = useState(true);
  const [userEnabled, setUserEnabled] = useState(true);
  const videoRefs = useRef<{[key: number]: HTMLVideoElement}>({});

  useEffect(() => {
    if (userEnabled) {
      fetchVideos();
    }
  }, [productId, userEnabled]);

  const fetchVideos = async () => {
    setLoading(true);
    try {
      const response = await fetch(`/api/videos/recommendations/${productId}?limit=5`);
      const data = await response.json();
      setVideos(data.recommendations);
    } catch (error) {
      console.error('Failed to fetch videos:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleVideoView = (videoId: number) => {
    // Track view event
    fetch('/api/videos/track', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        video_id: videoId,
        event_type: 'view',
        watch_duration: 0
      })
    });
  };

  const handleVideoPlay = (videoId: number) => {
    // Pause other videos
    Object.entries(videoRefs.current).forEach(([id, video]) => {
      if (parseInt(id) !== videoId && !video.paused) {
        video.pause();
      }
    });
  };

  const formatDuration = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins}:${secs.toString().padStart(2, '0')}`;
  };

  if (!userEnabled) {
    return (
      <div className={`video-sidebar ${position}`}>
        <div className="p-4 text-center">
          <p className="text-gray-600 mb-2">Video recommendations disabled</p>
          <button
            onClick={() => setUserEnabled(true)}
            className="text-blue-600 hover:underline"
          >
            Enable videos
          </button>
        </div>
      </div>
    );
  }

  if (loading) {
    return (
      <div className={`video-sidebar ${position} w-80 bg-gray-50 border-l border-gray-200`}>
        <div className="p-4">
          <div className="animate-pulse space-y-4">
            {[1, 2, 3].map((i) => (
              <div key={i} className="bg-gray-200 h-48 rounded"></div>
            ))}
          </div>
        </div>
      </div>
    );
  }

  if (videos.length === 0) {
    return null;
  }

  return (
    <div className={`video-sidebar ${position} w-80 bg-gray-50 border-l border-gray-200 overflow-y-auto`}>
      <div className="p-4">
        <div className="flex justify-between items-center mb-4">
          <h3 className="font-semibold text-lg">You May Also Like</h3>
          <button
            onClick={() => setUserEnabled(false)}
            className="text-gray-400 hover:text-gray-600"
            title="Disable videos"
          >
            ✕
          </button>
        </div>

        <div className="space-y-4">
          {videos.map((video) => (
            <div key={video.id} className="video-card bg-white rounded-lg shadow overflow-hidden">
              {/* Video Player */}
              <div className="relative aspect-video">
                <video
                  ref={(el) => { if (el) videoRefs.current[video.id] = el; }}
                  src={video.video_url}
                  poster={video.thumbnail_url}
                  className="w-full h-full object-cover"
                  muted
                  loop
                  playsInline
                  preload="metadata"
                  onPlay={() => handleVideoPlay(video.id)}
                  onLoadedMetadata={() => handleVideoView(video.id)}
                  controls={false}
                  onClick={(e) => {
                    const video = e.currentTarget;
                    if (video.paused) {
                      video.play();
                      video.controls = true;
                    } else {
                      video.pause();
                    }
                  }}
                />
                
                {/* Duration Badge */}
                <div className="absolute bottom-2 right-2 bg-black bg-opacity-75 text-white text-xs px-2 py-1 rounded">
                  {formatDuration(video.duration)}
                </div>

                {/* Play Overlay */}
                <div className="absolute inset-0 flex items-center justify-center bg-black bg-opacity-0 hover:bg-opacity-30 transition-all cursor-pointer">
                  <div className="w-12 h-12 bg-white bg-opacity-90 rounded-full flex items-center justify-center">
                    ▶
                  </div>
                </div>
              </div>

              {/* Video Info */}
              <div className="p-3">
                <h4 className="font-medium text-sm line-clamp-2 mb-1">
                  {video.title || video.product_name}
                </h4>
                
                <div className="flex justify-between items-center text-xs text-gray-600">
                  <span>{video.views_count.toLocaleString()} views</span>
                  <span className="capitalize">{video.video_type}</span>
                </div>

                {/* Product Price */}
                {video.product_price && (
                  <div className="mt-2 font-semibold text-blue-600">
                    ${video.product_price.toFixed(2)}
                  </div>
                )}

                {/* Quick Actions */}
                <div className="mt-3 flex gap-2">
                  <button
                    onClick={() => window.location.href = `/products/${video.product_id}`}
                    className="flex-1 bg-blue-600 text-white text-xs py-2 rounded hover:bg-blue-700 transition"
                  >
                    View Product
                  </button>
                  <button
                    onClick={() => {/* Add to cart logic */}}
                    className="flex-1 bg-gray-100 text-gray-700 text-xs py-2 rounded hover:bg-gray-200 transition"
                  >
                    Add to Cart
                  </button>
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
};
```
